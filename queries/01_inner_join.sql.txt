-- Retrieve completed sales records with matching customer, product, and branch details
SELECT 
    s.sale_id,
    c.customer_name,
    p.product_name,
    b.branch_name,
    s.quantity,
    s.total_amount,
    s.sale_date
FROM sales s
INNER JOIN customers c ON s.customer_id = c.customer_id
INNER JOIN products p ON s.product_id = p.product_id
INNER JOIN branches b ON s.branch_id = b.branch_id;
